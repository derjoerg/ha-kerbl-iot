# No version cap on syrupy/pytest-cov: pytest-homeassistant-custom-component
# pins both of these to an exact version matching whatever Home Assistant
# core release it wraps, and that exact pin moves with every one of its
# releases. Any range we pick here eventually stops matching the currently
# published pytest-homeassistant-custom-component versions and makes the
# whole install unsatisfiable -- let it dictate the version instead.
syrupy
pytest-cov

# pytest-homeassistant-custom-component pulls in Home Assistant core, which
# imports hass-nabucasa -> acme at collection time even though this
# integration never touches either. Unpinned, pip resolves the latest
# josepy, which dropped the ComparableX509 compat class that the acme
# version bundled here still relies on, breaking test collection with
# "AttributeError: module 'josepy' has no attribute 'ComparableX509'".
# Pin to the last josepy release that still provides it.
josepy<2
